Helpers puros em `scripts/metrics/lib.mjs` testáveis com fixture.

**Tech Stack:** Node 22 ESM (`import.meta.filename`), `gh` CLI autenticado, vitest, JSONL.

export const CATEGORY_PREFIXES = {
  feat: "feature", fix: "bugfix", docs: "docs", chore: "chore", refactor: "refactor",
};
export function categoryFromBranch(branch) {
  const prefix = (branch || "").split("/")[0];
  return CATEGORY_PREFIXES[prefix] ?? "outro";
}
export function firstGreenPrLocal(events, branch) {
  const scoped = events.filter((e) => (e.branch ?? e.data?.branch) === branch);
  const prePrs = scoped.filter((e) => e.type === "pre-pr");
  const firstPass = prePrs.find((e) => e.errors === 0 || e.description?.includes("PASS") || e.data?.result === "PASS");
  if (!firstPass) return { green: false, firstPassAt: null };
  const firstPassIdx = scoped.indexOf(firstPass);
  const earlierFails = scoped.slice(0, firstPassIdx).filter((e) => e.type === "rule:fail");
  return { green: earlierFails.length === 0, firstPassAt: firstPass.timestamp };
}
export function proxyBypassPr(pr, events) {
  const branch = pr.branch ?? pr.headRefName ?? "";
  if (!branch || pr.state !== "MERGED") return false;
  return !events.some((e) => e.type === "pre-pr" && (e.branch ?? e.data?.branch) === branch);
}
export const E2E_SKIP_BY_DESIGN = new Set(["e2e-smoke", "e2e-smoke-prod", "e2e-full"]);
export function isSkipByDesign(name) { return E2E_SKIP_BY_DESIGN.has(name); }
export function splitByModel(events) {
  const map = new Map();
  for (const e of events) {
    const model = e.model ?? e.resolvedModel ?? "sem-modelo";
    if (!map.has(model)) map.set(model, []);
    map.get(model).push(e);
  }
  return [...map.entries()].map(([model, events2]) => ({ model, events: events2 }));
}
```

import { execSync } from "child_process";
function ghApi(path, jq = ".") {
  const cmd = `gh api --paginate "${path}" --jq '${jq}'`;
  return JSON.parse(execSync(cmd, { encoding: "utf8", maxBuffer: 64 * 1024 * 1024 }));
}
export async function collectGhData({ since = "" } = {}) {
  const prs = ghApi("repos/andre0787/mileage-flow-manager/pulls?state=all", "[.[] | {number, title, state, createdAt, closedAt, mergedAt, headRefName}]");
  const filtered = since ? prs.filter((p) => p.createdAt >= since) : prs;
  const checksByPr = {};
  for (const pr of filtered) {
    const sha = ghApi(`repos/andre0787/mileage-flow-manager/pulls/${pr.number}`, ".head.sha");
    const checks = ghApi(`repos/andre0787/mileage-flow-manager/commits/${sha}/check-runs`, "[.check_runs[] | {name, conclusion, status, completedAt}]");
    checksByPr[pr.number] = checks;
  }
  return { prs: filtered, checksByPr };
}
```

```js
import { readFileSync } from "fs";
import { resolve } from "path";
import { parseProcessEvents } from "../lib/process-events.mjs";
import { computeCycleTime, isPrePrFail, isPrePrPass } from "../kpi-report.mjs";
import { categoryFromBranch, firstGreenPrLocal, proxyBypassPr, isSkipByDesign, splitByModel } from "./lib.mjs";
export function loadLocalEvents() {
  const p = resolve(import.meta.dirname, "../../docs/tracking/events.jsonl");
  return parseProcessEvents(readFileSync(p, "utf8"));
}
export function aggregate(gh, events) {
  const { prs, checksByPr } = gh;
  const byCategory = {};
  const greenFirst = { total: 0, green: 0 };
  const rework = { prsReopened: 0, prePrRepeated: 0 };
  const e2eSkips = { total: 0, byDesign: 0 };
  const bypassProxy = [];
  for (const pr of prs) {
    const cat = categoryFromBranch(pr.headRefName ?? "");
    byCategory[cat] = (byCategory[cat] ?? 0) + 1;
    const local = firstGreenPrLocal(events, pr.headRefName ?? "");
    if (pr.state === "MERGED") {
      greenFirst.total += 1;
      if (local.green) greenFirst.green += 1;
    }
    if (proxyBypassPr(pr, events)) bypassProxy.push(pr.number);
    const checks = checksByPr[pr.number] ?? [];
    for (const c of checks) {
      if (c.conclusion === "skipped") {
        e2eSkips.total += 1;
        if (isSkipByDesign(c.name)) e2eSkips.byDesign += 1;
      }
    }
    if (pr.number && !pr.state && false) rework.prsReopened += 1;
  }
  const prePrFails = events.filter((e) => e.type === "pre-pr" && isPrePrFail(e));
  rework.prePrRepeated = prePrFails.length;
  const cycleHours = computeCycleTime(events);
  const byModel = splitByModel(events);
  return { byCategory, greenFirst, cycleHours, rework, e2eSkips, bypassProxy, byModel };
}
```

```js
import { mkdirSync, writeFileSync } from "fs";
import { resolve } from "path";
export function writeReport(result, dateLabel) {
  const dir = resolve(import.meta.dirname, "../../docs/metrics");
  mkdirSync(dir, { recursive: true });
  const md = [
    `# Métricas do Programa — ${dateLabel}`,
    "",
    `## Verde-na-1ª: ${result.greenFirst.green}/${result.greenFirst.total}`,
    `## Tempo até verificação (médio): ${result.cycleHours ?? "n/d"}h`,
    `## Falhas por categoria: ${JSON.stringify(result.byCategory)}`,
    `## Retrabalho: ${JSON.stringify(result.rework)}`,
    `## Skips E2E: ${result.e2eSkips.total} (${result.e2eSkips.byDesign} por design)`,
    `## Bypasses (proxy, PRs sem pre-pr): ${result.bypassProxy.length}`,
  ].join("\n");
  const jsonPath = resolve(dir, `${dateLabel}-metrics.json`);
  writeFileSync(resolve(dir, `${dateLabel}-metrics.md`), md, "utf8");
  writeFileSync(jsonPath, JSON.stringify({ generatedAt: new Date().toISOString(), ...result }, null, 2), "utf8");
  return { md, jsonPath };
}
```
